\subsection{Tìm kiếm Cây Monte Carlo với UCT (MCTS-UCT)}
\label{subsec:mcts_uct}

\textit{[Phần của TV2 -- P5.2a: Trình bày 4 pha của MCTS (Selection, Expansion, Simulation, Backpropagation với quy tắc đổi dấu theo góc nhìn người vừa đánh), công thức UCT:}
\[
\text{UCT}(v_i) = \frac{W_i}{N_i} + c \sqrt{\frac{\ln N}{N_i}},
\]
\textit{mã giả (pseudocode), 3 tiêu chuẩn chọn nước đi cuối tại gốc (Robust Child $\arg\max N_i$, Max Child $\arg\max W_i/N_i$, Max-Robust), cùng các biến thể và cải tiến UCT (B2, B3).]}
